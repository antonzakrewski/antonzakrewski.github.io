distributivgesetz der doppelten negation einer Implikation unter EFQ

Satz

Innerhalb der intuitionistische Logik gilt

\begin{align*} (\lnot \lont A \rightarrow \lnot\lnot B) \rightarrow \lnot\lnot (A \rightarrow B) \end{align*}

Beweis

Ausschreiben der negation ergibt

\begin{align*} (((A \rightarrow \bot) \rightarrow \bot) \rightarrow ((B \rightarrow \bot) \rightarrow \bot)) \rightarrow (((A \rightarrow B) \rightarrow \bot) \rightarrow \bot) \end{align*}
\begin{prooftree}
\hypo{[z: $(A \rightarrow B) \rightarrow \bot$]}

\hypo{u: [$\bot \rightarrow B$]}

\hypo{[w: $((A \rightarrow \bot) \rightarrow \bot) \rightarrow (B \rightarrow \bot) \rightarrow \bot$]}

\hypo{[x: $A \rightarrow \bot$]}
\hypo{[y: A]}
\infer2[ $\rightarrow^-$ ]{$\bot$}
\infer1[ $\rightarrow^+ $x]{$(A \rightarrow \bot) \rightarrow \bot$}

\infer2[ $\rightarrow^-$]{$(B \rightarrow \bot) \rightarrow \bot$}

\hypo{[z: $(A \rightarrow B) \rightarrow \bot$]}
\hypo{[s: B]}
\infer1[ $\rightarrow^+$ y]{$A \rightarrow B$} 
\infer2[ $\rightarrow^-$]{$\bot$} 
\infer1[ $\rightarrow^+$ s]{$B \rightarrow \bot$}
\infer2[ $\rightarrow^-$]{$\bot$}
\infer2[ $\rightarrow^-$]{B}
\infer1[ $\rightarrow^+$ y]{$ A \rightarrow B$}
\infer2[ $\rightarrow^-$]{$\bot$}
\infer1[ $\rightarrow^+$ z]{$((A \rightarrow \bot) \rightarrow \bot) \rightarrow \bot$}
\infer1[ $\rightarrow^+$ w]{$(((A \rightarrow \bot) \rightarrow \bot) \rightarrow (B \rightarrow \bot) \rightarrow \bot) \rightarrow  ((A \rightarrow \bot) \rightarrow \bot) \rightarrow \bot$}
\infer1[ $\rightarrow^+$ u]{$(\bot \rightarrow B) \rightarrow ((((A \rightarrow \bot) \rightarrow \bot) \rightarrow (B \rightarrow \bot) \rightarrow \bot) \rightarrow  ((A \rightarrow \bot) \rightarrow \bot) \rightarrow \bot)$}
\end{prooftree}
