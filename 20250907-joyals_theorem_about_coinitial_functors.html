\begin{align*} \mathrm{map}_{\mathrm{Cat}_{\infty}/\mathcal{J}}(\mathcal{I}, X) \rightarrow \mathrm{map}_{\mathrm{Cat}_{\infty}/\mathcal{J}}(\mathcal{J}, X) \end{align*}
