\begin{align*} H_n(\mathbb{R}^n, U;M) \rightarrow \prod_{x \in \mathbb{R}^n \setminus U} \end{align*}

\begin{align*} (\mathbb{R}^n, U) \rightarrow (\mathbb{R}^n, \mathbb{R}^n \setminus \{x\}) \end{align*}
