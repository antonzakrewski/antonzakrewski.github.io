\begin{align*} \pi_{1}(M \# N) \cong \begin{cases} \pi_1(M) \coprod \pi_1(N) & \mbox{if } n > 2 \\ \pi_1(M) \coprod_{\mathbb{Z}} \pi_1(N) & \mbox{if } n = 2 \\ \end{cases} \end{align*}
