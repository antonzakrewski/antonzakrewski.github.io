\begin{align*} \mathrm{Grp}(-): \mathrm{Pr}^L \rightarrow& \mathrm{Pr}^L \\ \mathcal{C} \mapsto& \mathrm{Grp}(\mathcal{C}) \end{align*}

\begin{align*} \mathcal{C} \mapsto \mathcal{C} \otimes \mathrm{Mon}(\mathrm{An}) \simeq \mathrm{Mon}(\mathcal{C}) \end{align*}
