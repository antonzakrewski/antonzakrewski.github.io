\begin{align*} \pi_*(I_{\mathbb{Z}}) \cong& \begin{cases} 0 & \mbox{if } * > 0 \\ \mathbb{Z} & \mbox{if } * = 0 \\ 0 & \mbox{if } & \mbox{if } * = -1 \\ \\ \pi_{-(*+1)}(\mathbb{S}) & \mbox{if } * \leq -2 \\ \end{cases} \end{align*}
