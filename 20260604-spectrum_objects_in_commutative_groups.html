\begin{align*} \mathrm{Sp}(\mathrm{CGrp}(\mathcal{C})) \simeq& \mathrm{CGrp}(\mathrm{Sp}(\mathcal{C})) \\ \simeq& \mathrm{Sp}(\mathcal{C}) \end{align*}
