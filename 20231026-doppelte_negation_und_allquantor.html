doppelte Negation und Allquantor

Satz

innerhalb der minimal Logik gilt

\begin{align*} \vdash \lnot\lnot \forall_x A(x) \rightarrow \forall_x \lnot\lnot A(x) \end{align*}

Beweis

Ausschreiben der Negation ergibt

\begin{align*} (((\forall_x A(x)) \rightarrow \bot ) \rightarrow \bot ) \rightarrow (\forall_x (A(x) \rightarrow \bot) \rightarrow \bot) \end{align*}

intuitionistische einführung einer doppelten Negation

Es folgt 

\begin{prooftree}

\hypo{[1: $(\forall_x A(x) \rightarrow \bot) \rightarrow F$]}
\hypo{[2: $\forall_x A(x) \rightarrow \bot$]}
\infer2[ $\rightarrow^-$]{$\bot$}
\infer1[ $\rightarrow^+$ 3]{ $\forall_x A(x) \rightarrow \bot$}

\hypo{[2: $\forall_x A(x) \rightarrow \bot$]}
\hypo{[3: $\forall_x A(x)$]}
\infer2[ $\rightarrow^-$]{$\bot$} 
\infer1[ $\rightarrow^+$ 2]{ $(\forall_x A(x) \rightarrow \bot) \rightarrow F$}


\hypo{[3: $\forall_x A(x)$]}
\hypo{t}
\infer2[ $\forall^-$]{ A(t)}
\ellipsis{M}{$(A(t) \rightarrow \bot)  \rightarrow F$}
\infer1[$ \forall^+ \text{var. cond.}$]{$\forall_x (A(x) \rightarrow \bot) \rightarrow F$}

\infer2[ $\land^+$]{$(\forall_x (A(x) \rightarrow \bot) \rightarrow F) \land (\forall_x A(x) \rightarrow F) \rightarrow F$}
\infer1[ $\land_{2}^-$]{$\forall_x (A(x) \rightarrow \bot) \rightarrow F)$}
\infer1[ $\rightarrow^+$ 1]{$((\forall_x A(x) \rightarrow \bot) \rightarrow F) \rightarrow (\forall_x (A(x) \rightarrow F) \rightarrow F)$}
\end{prooftree}
