\begin{align*} \mathrm{lim}_{n \rightarrow \infty}( \mathbb{P}(a \leq \frac{S_n - np}{\sqrt{npq}} \leq b) = \frac{1}{\sqrt{2\pi}}\int_{a}^b e^{- \frac{x^2}{2}} \mathrm{d}x \end{align*}
